# file structure

Every tracked file in the repo. Build output, `node_modules` and lockfiles are
omitted.

```sh
kp-app/
├── .env                            # one file; every service reads it
├── .env.example
├── .dockerignore
├── .gitignore
├── .prettierignore                 # Prettier is shared from the root
├── .prettierrc
├── AGENTS.md                       # the doc index every agent reads first
├── CLAUDE.md -> AGENTS.md          # symlink, so one file serves every harness
├── README.md
├── Makefile                        # make up / down / migrate / revision / logs
├── docker-compose.yml              # db + server + web
├── package.json                    # root scripts — each one a `turbo run`
├── pnpm-workspace.yaml             # workspaces: apps/* and packages/*
├── turbo.json                      # task graph and cache config
├── scripts/                        # repo-level scripts (empty)
│
├── apps/                           # deployables — nothing imports these
│   ├── README.md
│   │
│   ├── server/                     # NestJS 12 (ESM) + Drizzle + Postgres  :8000
│   │   ├── README.md
│   │   ├── package.json
│   │   ├── Dockerfile.dev
│   │   ├── nest-cli.json
│   │   ├── oxlint.json
│   │   ├── drizzle.config.ts       # points drizzle-kit at src/db/schema.ts
│   │   ├── tsconfig.json
│   │   ├── tsconfig.build.json
│   │   ├── vitest.config.ts        # unit specs — src/**/*.spec.ts
│   │   ├── vitest.config.e2e.ts    # e2e specs — test/*.e2e-spec.ts
│   │   ├── drizzle/                # generated migrations, checked in
│   │   │   ├── 0000_add_users_table.sql
│   │   │   └── meta/
│   │   │       ├── _journal.json
│   │   │       └── 0000_snapshot.json
│   │   ├── src/
│   │   │   ├── main.ts             # bootstrap, CORS, shutdown hooks, listen
│   │   │   ├── setup-app.ts        # /api prefix + versioning; used by e2e too
│   │   │   ├── app.module.ts       # env, global validation pipe, module list
│   │   │   ├── config/             # infrastructure, shared by every module
│   │   │   │   ├── env.ts            Zod-validated environment
│   │   │   │   └── database-url.ts   URL scheme normalizing
│   │   │   ├── db/                 # infrastructure, shared by every module
│   │   │   │   ├── db.module.ts      @Global — provides the `DB` token
│   │   │   │   └── schema.ts         Drizzle tables; migrations diff this
│   │   │   └── modules/            # one folder per API slice
│   │   │       ├── health/
│   │   │       │   ├── health.controller.ts       GET /health, unversioned
│   │   │       │   ├── health.controller.spec.ts
│   │   │       │   └── health.module.ts
│   │   │       └── users/          # the worked example: CRUD over one table
│   │   │           ├── users.controller.ts
│   │   │           ├── users.controller.spec.ts
│   │   │           ├── users.service.ts
│   │   │           └── users.module.ts
│   │   └── test/                   # e2e — boots the app against a real database
│   │       ├── health.e2e-spec.ts
│   │       └── users.e2e-spec.ts
│   │
│   ├── web/                        # React 19 + Vite 8 + React Router 8    :5173
│   │   ├── README.md
│   │   ├── package.json
│   │   ├── Dockerfile.dev
│   │   ├── index.html
│   │   ├── .oxlintrc.json
│   │   ├── .prettierignore
│   │   ├── .gitignore
│   │   ├── vite.config.ts          # React + Tailwind plugins, @ alias, envDir
│   │   ├── vitest.config.ts        # jsdom, merged onto vite.config.ts
│   │   ├── vitest.setup.ts         # re-exports @kp-app/ui/vitest-setup
│   │   ├── tsconfig.json
│   │   ├── tsconfig.app.json       # @/* -> ./src/*
│   │   ├── tsconfig.node.json
│   │   ├── public/
│   │   │   └── favicon.svg
│   │   └── src/
│   │       ├── main.tsx            # createRoot, TooltipProvider, router
│   │       ├── App.tsx             # layout shell, renders routes via <Outlet />
│   │       ├── index.css           # one line: @import '@kp-app/ui/styles.css'
│   │       ├── app/                # the pages
│   │       │   ├── routes.ts         the page list: routes, nav and titles
│   │       │   ├── home/
│   │       │   │   └── Page.tsx      one default export per page
│   │       │   └── about/
│   │       │       └── Page.tsx
│   │       ├── components/
│   │       │   ├── README.md
│   │       │   └── common/         # pieces several pages share (empty)
│   │       └── lib/
│   │           ├── api.ts            axios client + contract parsing
│   │           ├── api.test.ts
│   │           └── site.ts           product-wide strings
│   │
│   ├── landing/                    # Next.js 16, App Router                :3000
│   │   ├── README.md
│   │   ├── package.json
│   │   ├── AGENTS.md               # generated by `next dev`, rewritten each run
│   │   ├── CLAUDE.md               # one line: @AGENTS.md
│   │   ├── next.config.ts          # transpilePackages: ['@kp-app/ui']
│   │   ├── next-env.d.ts
│   │   ├── postcss.config.mjs      # Tailwind via PostCSS, not the Vite plugin
│   │   ├── .oxlintrc.json
│   │   ├── .prettierignore
│   │   ├── .gitignore
│   │   ├── tsconfig.json
│   │   └── src/app/
│   │       ├── globals.css         # the same one-line @import as web
│   │       ├── layout.tsx          # <html>/<body> shell and metadata
│   │       └── page.tsx
│   │
│   ├── mcp-app/                    # placeholder, empty
│   └── mobile/                     # placeholder, empty
│
├── packages/                       # shared code — imported by apps
│   ├── README.md
│   │
│   ├── shared/                     # the API contract; compiles to dist/
│   │   ├── README.md
│   │   ├── package.json
│   │   ├── tsconfig.json
│   │   └── src/
│   │       ├── index.ts            # barrel over all three folders
│   │       ├── constants/          # resource names and paths — depends on nothing
│   │       │   ├── index.ts
│   │       │   ├── http.ts           the /api/v1 prefix
│   │       │   ├── health.ts
│   │       │   └── users.ts
│   │       ├── schemas/            # the Zod schemas, which read the constants
│   │       │   ├── index.ts
│   │       │   ├── http.ts           the error body every endpoint can return
│   │       │   ├── health.ts
│   │       │   └── users.ts
│   │       └── types/              # the types inferred from those schemas
│   │           ├── index.ts
│   │           ├── http.ts
│   │           ├── health.ts
│   │           └── users.ts
│   │
│   └── ui/                         # tokens + components; ships source, no build
│       ├── README.md
│       ├── package.json
│       ├── components.json         # shadcn CLI settings, reused by ui:add
│       ├── vite.config.ts
│       ├── vitest.config.ts
│       ├── tsconfig.json
│       ├── tsconfig.lib.json
│       ├── tsconfig.node.json
│       ├── .oxlintrc.json
│       ├── .prettierignore
│       ├── .gitignore
│       ├── .storybook/
│       │   ├── main.ts
│       │   └── preview.tsx         # loads the tokens; dark-mode toolbar switch
│       ├── scripts/
│       │   ├── lint-tokens.ts      # the token linter; every app runs it
│       │   └── ui-add.ts           # `ui:add` — shadcn CLI plus a reshape
│       ├── testing/
│       │   └── vitest-setup.ts     # DOM stubs Radix needs; shared with apps
│       └── src/
│           ├── index.ts            # public surface: every primitive, plus `cn`
│           ├── tokens/
│           │   ├── README.md
│           │   ├── index.css         the stylesheet every app imports
│           │   ├── primitives.css    raw values, referencing nothing
│           │   ├── semantics.css     roles, each pointing at one primitive
│           │   ├── theme.css         which tokens become utility classes

│           │   ├── tokens.test.ts
│           │   └── Tokens.stories.tsx
│           └── components/
│               ├── composites/     # built from primitives, app-agnostic (empty)
│               └── primitives/     # shadcn building blocks
│                   ├── README.md
│                   ├── index.ts      barrel over every folder below
│                   ├── Button/       the shape every primitive follows
│                   │   ├── Button.tsx
│                   │   ├── Button.stories.tsx
│                   │   ├── Button.test.tsx
│                   │   └── index.ts
│                   └── …             same four files each: Accordion, Alert,
│                                     AlertDialog, Avatar, Badge, Card, Checkbox,
│                                     Dialog, DropdownMenu, Input, Label, Popover,

│
└── docs/
```
